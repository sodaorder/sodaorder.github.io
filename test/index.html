// 시트명 정의 const SHEET = { ORDER: "주문내역DB", SUBMIT: "폼DB", CHANNEL: "채널ID" }; const API_KEY = "soda_sec_key_t0d72x9z"; // ----------------------------------------------------------------- // 1. GET 요청 처리 (웹페이지 접속 & VBA TSV 수신) // ----------------------------------------------------------------- function doGet(e) { try { const params = (e && e.parameter) ? e.parameter : {}; const action = params.action || ""; const token = params.t || params.token || ""; // [VBA 전용] 폼DB 전체 텍스트 수신 (GET) if (action === "FETCH_FORM_DB") { if (params.apiKey !== API_KEY) { return ContentService.createTextOutput("ERROR: Invalid API Key"); } const ss = SpreadsheetApp.getActiveSpreadsheet(); const submitSheet = ss.getSheetByName(SHEET.SUBMIT); if (!submitSheet) return ContentService.createTextOutput("ERROR: SHEET NOT FOUND"); const data = submitSheet.getDataRange().getValues(); const tsv = data.map(row => row.map(cell => String(cell).replace(/\r?\n|\r/g, " ")).join("\t") ).join("\n"); return ContentService.createTextOutput(tsv).setMimeType(ContentService.MimeType.TEXT); } // [공개 API 1] 고객 웹페이지 - 주문 정보 조회 if (action === "GET_ORDER_DATA") { return createJsonResponse(getOrderData(token)); } // [공개 API 2] 고객 웹페이지 - 본인 인증 및 기존 배송지 조회 if (action === "VERIFY_AUTH") { return createJsonResponse(verifyAuth(token, params.authCode)); } return ContentService.createTextOutput("GAS Backend API Server is Running."); } catch (err) { return createJsonResponse({ success: false, message: "GET 서버 오류: " + err.toString() }); } } // ----------------------------------------------------------------- // 2. POST 요청 처리 (웹페이지 폼 저장 & 엑셀 업로드) // ----------------------------------------------------------------- function doPost(e) { try { let contents = {}; if (e && e.postData && e.postData.contents) { try { contents = JSON.parse(e.postData.contents); } catch (jsonErr) { contents = {}; } } const action = contents.action || ""; // [공개 API 3] 고객 배송지 정보 저장 (SAVE_SHIPPING) if (action === "SAVE_SHIPPING") { return createJsonResponse(saveShipping(contents)); } // [관리자 API] 엑셀 전용 요청 (API Key 필수 검증) if (contents.apiKey !== API_KEY) { return createJsonResponse({ success: false, message: "인증 실패: 유효하지 않은 API 키입니다." }); } if (action === "UPLOAD_ORDERS") { return createJsonResponse(handleUploadOrders(contents.orders || [])); } else if (action === "GET_ALL_SHIPPING_DATA") { return createJsonResponse(getAllShippingData()); } return createJsonResponse({ success: false, message: "알 수 없는 요청(Action)입니다: " + action }); } catch (err) { return createJsonResponse({ success: false, message: "POST 서버 오류: " + err.toString() }); } } // ----------------------------------------------------------------- // 3. [공개 API] GET_ORDER_DATA (주문 기본 정보 + 계좌 정보 수신) // ----------------------------------------------------------------- function getOrderData(token) { if (!token) { return { success: false, message: "유효하지 않은 접근입니다. [공개 API] VERIFY_AUTH (본인 인증 및 제출 내역 확인) // ----------------------------------------------------------------- function verifyAuth(token, authCode) { if (!token) { return { success: false, message: "토큰이 누락되었습니다." }; } const ss = SpreadsheetApp.getActiveSpreadsheet(); const submitSheet = ss.getSheetByName(SHEET.SUBMIT); let existingAddress = null; if (submitSheet) { const submitCol = getColumnIndexMap(submitSheet); const submitData = submitSheet.getDataRange().getValues(); for (let i = 1; i < submitData.length; i++) { const dbToken = String(submitData[i][submitCol["토큰"]] || "").trim(); if (dbToken !== "" && dbToken === String(token).trim()) { existingAddress = { receiverName: String(submitData[i][submitCol["수령인성함"]] || ""), phone: String(submitData[i][submitCol["연락처"]] || ""), postcode: String(submitData[i][submitCol["우편번호"]] || ""), addressBase: String(submitData[i][submitCol["기본주소"]] || ""), addressDetail: String(submitData[i][submitCol["상세주소"]] || ""), memo: String(submitData[i][submitCol["배송요청사항"]] || ""), depositorName: String(submitData[i][submitCol["입금자명"]] || "") }; break; } } } return { success: true, address: existingAddress }; } // ----------------------------------------------------------------- // 5. [공개 API] SAVE_SHIPPING (배송지 정보 저장) // ----------------------------------------------------------------- function saveShipping(formData) { const ss = SpreadsheetApp.getActiveSpreadsheet(); const submitSheet = ss.getSheetByName(SHEET.SUBMIT); const orderSheet = ss.getSheetByName(SHEET.ORDER); const submitCol = getColumnIndexMap(submitSheet); const orderCol = getColumnIndexMap(orderSheet); const nowStr = Utilities.formatDate(new Date(), "Asia/Seoul", "yyyy-MM-dd HH:mm:ss"); const tokenVal = String(formData.token || "").trim(); if (!tokenVal) { return { success: false, message: "토큰 정보가 유효하지 않습니다." }; } let targetChannel = ""; const orderData = orderSheet.getDataRange().getValues(); for (let i = 1; i < orderData.length; i++) { if (String(orderData[i][orderCol["토큰"]]).trim() === tokenVal) { targetChannel = String(orderData[i][orderCol["채널명"]]).trim(); break; } } const submitData = submitSheet.getDataRange().getValues(); let existingRowIndex = -1; for (let i = 1; i < submitData.length; i++) { if (String(submitData[i][submitCol["토큰"]]).trim() === tokenVal) { existingRowIndex = i; break; } } const numCols = submitSheet.getLastColumn(); if (existingRowIndex !== -1) { const targetRow = submitData[existingRowIndex]; if (submitCol["제출일시"] !== undefined) targetRow[submitCol["제출일시"]] = nowStr; if (submitCol["채널명"] !== undefined) targetRow[submitCol["채널명"]] = targetChannel; if (submitCol["닉네임"] !== undefined) targetRow[submitCol["닉네임"]] = formData.nickname || ""; if (submitCol["인증번호"] !== undefined) targetRow[submitCol["인증번호"]] = formData.authNo || ""; if (submitCol["수령인성함"] !== undefined) targetRow[submitCol["수령인성함"]] = formData.receiverName || ""; if (submitCol["연락처"] !== undefined) targetRow[submitCol["연락처"]] = formData.phone || ""; if (submitCol["우편번호"] !== undefined) targetRow[submitCol["우편번호"]] = formData.postcode || ""; if (submitCol["기본주소"] !== undefined) targetRow[submitCol["기본주소"]] = formData.addressBase || ""; if (submitCol["상세주소"] !== undefined) targetRow[submitCol["상세주소"]] = formData.addressDetail || ""; if (submitCol["배송요청사항"] !== undefined) targetRow[submitCol["배송요청사항"]] = formData.memo || ""; if (submitCol["입금자명"] !== undefined) targetRow[submitCol["입금자명"]] = formData.depositorName || ""; submitSheet.getRange(existingRowIndex + 1, 1, 1, targetRow.length).setValues([targetRow]); } else { const newRow = new Array(numCols).fill(""); if (submitCol["토큰"] !== undefined) newRow[submitCol["토큰"]] = tokenVal; if (submitCol["제출일시"] !== undefined) newRow[submitCol["제출일시"]] = nowStr; if (submitCol["채널명"] !== undefined) newRow[submitCol["채널명"]] = targetChannel; if (submitCol["닉네임"] !== undefined) newRow[submitCol["닉네임"]] = formData.nickname || ""; if (submitCol["인증번호"] !== undefined) newRow[submitCol["인증번호"]] = formData.authNo || ""; if (submitCol["수령인성함"] !== undefined) newRow[submitCol["수령인성함"]] = formData.receiverName || ""; if (submitCol["연락처"] !== undefined) newRow[submitCol["연락처"]] = formData.phone || ""; if (submitCol["우편번호"] !== undefined) newRow[submitCol["우편번호"]] = formData.postcode || ""; if (submitCol["기본주소"] !== undefined) newRow[submitCol["기본주소"]] = formData.addressBase || ""; if (submitCol["상세주소"] !== undefined) newRow[submitCol["상세주소"]] = formData.addressDetail || ""; if (submitCol["배송요청사항"] !== undefined) newRow[submitCol["배송요청사항"]] = formData.memo || ""; if (submitCol["입금자명"] !== undefined) newRow[submitCol["입금자명"]] = formData.depositorName || ""; submitSheet.appendRow(newRow); } // 주문 상태 업데이트 (3: 제출 완료) for (let i = 1; i < orderData.length; i++) { if (String(orderData[i][orderCol["토큰"]]).trim() === tokenVal) { if (orderCol["주문상태"] !== undefined) { const currentStatus = parseInt(orderData[i][orderCol["주문상태"]], 10) || 0; if (3 > currentStatus) orderSheet.getRange(i + 1, orderCol["주문상태"] + 1).setValue(3); } if (orderCol["최종갱신일시"] !== undefined) orderSheet.getRange(i + 1, orderCol["최종갱신일시"] + 1).setValue(nowStr); break; } } return { success: true, message: "배송 정보가 저장되었습니다." }; } // ----------------------------------------------------------------- // 6. [관리자 API] 주문 내역 업로드 (세분화 토큰 복구 지원) // ----------------------------------------------------------------- function handleUploadOrders(orders) { const ss = SpreadsheetApp.getActiveSpreadsheet(); const orderSheet = ss.getSheetByName(SHEET.ORDER); const orderCol = getColumnIndexMap(orderSheet); const orderData = orderSheet.getDataRange().getValues(); const nowStr = Utilities.formatDate(new Date(), "Asia/Seoul", "yyyy-MM-dd HH:mm:ss"); const restoredTokens = []; orders.forEach(order => { const chName = String(order.channelName || "").trim(); const oDate = formatDateOnly(order.orderDate); const nick = String(order.nickname || "").trim(); let token = String(order.token || "").trim(); let foundRowIndex = -1; if (token !== "") { for (let i = 1; i < orderData.length; i++) { if (String(orderData[i][orderCol["토큰"]]).trim() === token) { foundRowIndex = i; break; } } } if (foundRowIndex === -1 && nick !== "") { for (let i = 1; i < orderData.length; i++) { const dbCh = String(orderData[i][orderCol["채널명"]]).trim(); const dbDate = formatDateOnly(orderData[i][orderCol["주문일자"]]); const dbNick = String(orderData[i][orderCol["닉네임"]]).trim(); if (dbCh === chName && dbDate === oDate && dbNick === nick) { foundRowIndex = i; token = String(orderData[i][orderCol["토큰"]]).trim(); restoredTokens.push({ nickname: nick, token: token }); break; } } } if (foundRowIndex !== -1) { const rowNum = foundRowIndex + 1; if (orderCol["채널명"] !== undefined) orderSheet.getRange(rowNum, orderCol["채널명"] + 1).setValue(chName); if (orderCol["주문일자"] !== undefined) orderSheet.getRange(rowNum, orderCol["주문일자"] + 1).setValue(oDate); if (orderCol["토큰"] !== undefined) orderSheet.getRange(rowNum, orderCol["토큰"] + 1).setValue(token); if (orderCol["닉네임"] !== undefined) orderSheet.getRange(rowNum, orderCol["닉네임"] + 1).setValue(nick); if (orderCol["합배여부"] !== undefined) orderSheet.getRange(rowNum, orderCol["합배여부"] + 1).setValue(order.combined || ""); if (orderCol["주문금액"] !== undefined) orderSheet.getRange(rowNum, orderCol["주문금액"] + 1).setValue(order.amount || 0); if (orderCol["수량"] !== undefined) orderSheet.getRange(rowNum, orderCol["수량"] + 1).setValue(order.quantity || 0); if (orderCol["주문상태"] !== undefined) { const existingStatus = parseInt(orderData[foundRowIndex][orderCol["주문상태"]], 10) || 0; if (existingStatus < 3) { orderSheet.getRange(rowNum, orderCol["주문상태"] + 1).setValue(order.status); } } if (orderCol["카드결제"] !== undefined) orderSheet.getRange(rowNum, orderCol["카드결제"] + 1).setValue(order.cardPay || ""); if (orderCol["최종갱신일시"] !== undefined) orderSheet.getRange(rowNum, orderCol["최종갱신일시"] + 1).setValue(nowStr); } else { const numCols = orderSheet.getLastColumn(); const newRow = new Array(numCols).fill(""); if (orderCol["채널명"] !== undefined) newRow[orderCol["채널명"]] = chName; if (orderCol["주문일자"] !== undefined) newRow[orderCol["주문일자"]] = oDate; if (orderCol["토큰"] !== undefined) newRow[orderCol["토큰"]] = token; if (orderCol["닉네임"] !== undefined) newRow[orderCol["닉네임"]] = nick; if (orderCol["합배여부"] !== undefined) newRow[orderCol["합배여부"]] = order.combined || ""; if (orderCol["주문금액"] !== undefined) newRow[orderCol["주문금액"]] = order.amount || 0; if (orderCol["수량"] !== undefined) newRow[orderCol["수량"]] = order.quantity || 0; if (orderCol["주문상태"] !== undefined) newRow[orderCol["주문상태"]] = order.status; if (orderCol["카드결제"] !== undefined) newRow[orderCol["카드결제"]] = order.cardPay || ""; if (orderCol["최종갱신일시"] !== undefined) newRow[orderCol["최종갱신일시"]] = nowStr; orderSheet.appendRow(newRow); } }); return { success: true, message: "주문 내역 업로드 완료", restoredTokens: restoredTokens }; } // ----------------------------------------------------------------- // 7. [관리자 API] 배송지 전체 데이터 조회 (엑셀 동기화용) // ----------------------------------------------------------------- function getAllShippingData() { const ss = SpreadsheetApp.getActiveSpreadsheet(); const submitSheet = ss.getSheetByName(SHEET.SUBMIT); if (!submitSheet) return { success: true, data: [] }; const submitCol = getColumnIndexMap(submitSheet); const data = submitSheet.getDataRange().getValues(); const list = []; for (let i = 1; i < data.length; i++) { const row = data[i]; const token = String(row[submitCol["토큰"]] || "").trim(); if (token !== "") { list.push({ token: token, receiverName: String(row[submitCol["수령인성함"]] || ""), phone: String(row[submitCol["연락처"]] || ""), postcode: String(row[submitCol["우편번호"]] || ""), addressBase: String(row[submitCol["기본주소"]] || ""), addressDetail: String(row[submitCol["상세주소"]] || ""), memo: String(row[submitCol["배송요청사항"]] || ""), depositorName: String(row[submitCol["입금자명"]] || "") }); } } return { success: true, data: list }; } // ----------------------------------------------------------------- // 헬퍼 함수: 계좌 정보 추출 (채널ID 시트 기준) // ----------------------------------------------------------------- function getBankInfo(channelName) { const ss = SpreadsheetApp.getActiveSpreadsheet(); const channelSheet = ss.getSheetByName(SHEET.CHANNEL); if (!channelSheet) { return { bankOwner: "-", accountNo: "-" }; } const colMap = getColumnIndexMap(channelSheet); const data = channelSheet.getDataRange().getValues(); for (let i = 1; i < data.length; i++) { const dbCh = String(data[i][colMap["채널명"]] || "").trim(); if (dbCh === channelName) { return { bankOwner: String(data[i][colMap["예금주"]] || "-"), accountNo: String(data[i][colMap["계좌번호"]] || "-") }; } } return { bankOwner: "-", accountNo: "-" }; } // ----------------------------------------------------------------- // 유틸리티 함수들 // ----------------------------------------------------------------- function getColumnIndexMap(sheet) { const headers = sheet.getRange(1, 1, 1, sheet.getLastColumn()).getValues()[0]; const map = {}; headers.forEach((h, idx) => { map[String(h).trim()] = idx; }); return map; } function formatDateOnly(val) { if (!val) return ""; if (val instanceof Date) { return Utilities.formatDate(val, "Asia/Seoul", "yyyy-MM-dd"); } const str = String(val).trim(); if (str.length >= 10) { return str.substring(0, 10).replace(/\//g, "-").replace(/\./g, "-"); } return str; } function createJsonResponse(obj) { return ContentService.createTextOutput(JSON.stringify(obj)) .setMimeType(ContentService.MimeType.JSON);
